		tag = NOR

		nwc_gov_alive = yes

	}

	option = {
		name = nwc_sweden.5.a
		if = {
			limit = { NOT = { has_idea = nwc_swe_advisor_1 } }
			add_ideas = nwc_swe_advisor_1

		else_if = {
			limit = { NOT = { has_idea = nwc_swe_advisor_2 } }
			add_ideas = nwc_swe_advisor_2
		}
		else = { add_ideas = nwc_swe_advisor_3 }
		set_temp_variable = { nwc_swe_delta = 8 }
		nwc_swe_opinion_add = yes
	}
	option = {
		name = nwc_sweden.5.b
		set_temp_variable = { nwc_swe_delta = -3 }
		nwc_swe_opinion_add = yes
	}
}
